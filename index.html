const SHEET_NAME = 'ChargeLog'; const HEADERS = ['id','date','time','odometer','battStart','battEnd','kwhAdded','cost','duration','location','chargerType']; // L:T — the sheet's own formula columns. Read-only: never written by addRow/updateRow. const CALC_HEADERS = ['range','kmPerPct','bahtPerPct','powerLoss','powerLossPct','dupCount','e20Price','e20Needed','e20Cost']; /** * Custom Function: Returns historical Thailand Gasohol E20 price for a given date. * If the exact date is not a price-adjustment day, it returns the price in effect on that day. * * @param {Date|string} dateInput The date to check (from column B). * @return {number|string} The E20 price in THB/liter. * @customfunction */ function GET_E20_PRICE(dateInput) { if (!dateInput) return ""; // Parse input date to YYYY-MM-DD var d = new Date(dateInput); if (isNaN(d.getTime())) return "Invalid Date"; var year = d.getFullYear(); var month = ("0" + (d.getMonth() + 1)).slice(-2); var day = ("0" + d.getDate()).slice(-2); var dateStr = year + "-" + month + "-" + day; // Key historical E20 price changes (Effective Date: Price in THB) // Source: EPPO / Retail Oil Archives (Bangchak / PTT) var priceHistory = [ { date: "2026-01-01", price: 34.68 }, { date: "2026-03-18", price: 34.68 }, { date: "2026-03-21", price: 32.68 }, { date: "2026-03-24", price: 34.68 }, { date: "2026-03-26", price: 40.68 }, { date: "2026-03-31", price: 41.68 }, { date: "2026-04-03", price: 38.95 }, { date: "2026-04-17", price: 35.95 }, { date: "2026-05-13", price: 36.35 }, { date: "2026-05-20", price: 37.90 }, { date: "2026-05-29", price: 37.90 }, { date: "2026-06-04", price: 38.30 }, { date: "2026-06-16", price: 41.93 }, { date: "2026-06-17", price: 39.98 }, { date: "2026-06-19", price: 38.85 }, { date: "2026-06-25", price: 37.68 }, { date: "2026-07-03", price: 37.08 }, { date: "2026-07-08", price: 34.57 }, { date: "2026-07-22", price: 35.42 }, { date: "2026-07-23", price: 36.32 }, { date: "2026-08-06", price: 35.62 }, { date: "2026-08-12", price: 36.47 }, { date: "2026-08-19", price: 37.32 }, { date: "2026-09-02", price: 37.92 }, { date: "2026-09-09", price: 38.72 }, { date: "2026-09-15", price: 39.57 }, { date: "2026-10-02", price: 40.32 }, { date: "2026-10-09", price: 41.07 }, { date: "2026-10-21", price: 41.67 }, { date: "2026-10-29", price: 41.17 }, { date: "2026-11-05", price: 40.47 }, { date: "2026-11-06", price: 39.47 } ]; // Find the latest price in effect on or before the given date var effectivePrice = null; for (var i = 0; i < priceHistory.length; i++) { if (priceHistory[i].date <= dateStr) { effectivePrice = priceHistory[i].price; } else { break; } } return effectivePrice !== null ? effectivePrice : "Before records"; } function doGet(e) { const p = e.parameter; try { switch(p.action) { case 'ping': return ok(ping()); case 'list': return ok(listRows()); case 'add': return ok(addRow(JSON.parse(p.data))); case 'update': return ok(updateRow(JSON.parse(p.data))); case 'delete': return ok(deleteRow(p.id)); default: return err('Unknown action'); } } catch(ex) { return err(ex.message); } } function ok(d) { return out({status:'ok',...d}); } function err(m) { return out({status:'error',message:m}); } function out(o) { return ContentService .createTextOutput(JSON.stringify(o)) .setMimeType(ContentService.MimeType.JSON); } function getSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sh = ss.getSheetByName(SHEET_NAME); if (!sh) { sh = ss.insertSheet(SHEET_NAME); sh.appendRow(HEADERS); sh.setFrozenRows(1); sh.getRange(1,1,1,HEADERS.length).setFontWeight('bold'); } return sh; } function ping() { const sh = getSheet(); return { rowCount: Math.max(0, sh.getLastRow()-1), sheetName: sh.getName() }; } function listRows() { const sh = getSheet(); const last = sh.getLastRow(); if (last < 2) return { data: [] }; const cols = HEADERS.length + CALC_HEADERS.length; // A:T const vals = sh.getRange(2,1,last-1,cols).getValues(); const tz = SpreadsheetApp.getActiveSpreadsheet().getSpreadsheetTimeZone(); const data = vals.filter(r => r[0]).map(r => { const o = {}; HEADERS.forEach((h,i) => { let v = r[i]; if (h === 'date' && v instanceof Date) v = Utilities.formatDate(v, tz, 'yyyy-MM-dd'); o[h] = v; }); CALC_HEADERS.forEach((h,i) => { o[h] = r[HEADERS.length+i]; }); o.odometer = Number(o.odometer)||0; o.battStart = Number(o.battStart)||0; o.battEnd = Number(o.battEnd)||0; o.kwhAdded = Number(o.kwhAdded)||0; o.duration = Number(o.duration)||0; o.cost = Number(o.cost)||0; o.range = Number(o.range)||0; o.kmPerPct = Number(o.kmPerPct)||0; o.bahtPerPct = Number(o.bahtPerPct)||0; o.powerLoss = Number(o.powerLoss)||0; o.powerLossPct = Number(o.powerLossPct)||0; o.e20Price = Number(o.e20Price)||0; o.e20Needed = Number(o.e20Needed)||0; o.e20Cost = Number(o.e20Cost)||0; return o; }); return { data }; } function findRow(id) { const sh = getSheet(); const last = sh.getLastRow(); if (last < 2) return -1; const ids = sh.getRange(2,1,last-1,1).getValues(); for (let i=0;i<ids.length;i++) if (ids[i][0]===id) return i+2; return -1; } function addRow(row) { const sh = getSheet(); sh.appendRow(HEADERS.map(h => row[h]!==undefined ? row[h] : '')); const newRow = sh.getLastRow(); // Columns L:T are formula columns outside HEADERS (A:K) — carry the // previous row's formulas down so the new row calculates too. if (newRow > 2) { sh.getRange(newRow - 1, 12, 1, 9).copyTo(sh.getRange(newRow, 12, 1, 9)); } return { added: row.id }; } function updateRow(row) { const r = findRow(row.id); if (r < 0) throw new Error('Row not found: '+row.id); getSheet().getRange(r,1,1,HEADERS.length) .setValues([HEADERS.map(h => row[h]!==undefined ? row[h] : '')]); return { updated: row.id }; } function deleteRow(id) { const r = findRow(id); if (r < 0) throw new Error('Row not found: '+id); getSheet().deleteRow(r); return { deleted: id }; }
